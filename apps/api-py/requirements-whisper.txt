# Optional dependencies for Local Whisper (on-device, private transcription).
#
# Not installed by default because the model + runtime are large. The app
# starts and runs fine without these; the Local Whisper provider simply reports
# itself unavailable and the onboarding prompts the user to enable it.
#
# Install:  pip install -r requirements-whisper.txt
#
# faster-whisper bundles CTranslate2 (CPU int8 works everywhere; GPU needs a
# matching CUDA runtime). numpy is used to feed buffered PCM windows from the
# live path into the model.
faster-whisper>=1.0.3
numpy>=1.26

# GPU acceleration (NVIDIA, Windows/Linux x86-64). CTranslate2 4.x needs CUDA 12
# cuBLAS + cuDNN 9 at runtime. Shipping them as pip wheels means a GPU machine
# gets ~30x faster STT (small model: ~13s on CPU -> ~0.3s on GPU) with zero
# manual CUDA setup; whisper_local_provider registers their DLL dirs at import.
# Harmless (just unused) on machines without an NVIDIA GPU — the provider falls
# back to CPU int8. ~1.2 GB of wheels, so install only where the GPU is wanted:
#   pip install nvidia-cublas-cu12 "nvidia-cudnn-cu12==9.*"
nvidia-cublas-cu12; platform_machine == "AMD64" or platform_machine == "x86_64"
nvidia-cudnn-cu12==9.*; platform_machine == "AMD64" or platform_machine == "x86_64"
